    <h1>🕹️ Free Arcade App</h1>
    <p>Install this web app via your browser settings to play offline!</p>

    <!-- SCREEN 1: MAIN MENU -->
    <div id="menu-screen" class="screen active">
        <h2>Ready to Play?</h2>
        <p>Watch a quick 5-second ad to unlock your daily game session.</p>
        <button onclick="startAd()">Watch Ad & Play</button>
    </div>

    <!-- SCREEN 2: THE AD GATE -->
    <div id="ad-screen" class="screen">
        <h2>Sponsor Message</h2>
        <div id="ad-box">
            <!-- PLACEHOLDER FOR AD NETWORK SCRIPT -->
            <p style="color: #aaa;">[Google AdSense / AdInPlay Video Ad Running]</p>
        </div>
        <div id="timer">You can skip in: <span id="countdown">5</span>s</div>
        <button id="skip-btn" onclick="startGame()" style="display:none; background:#ff3333; color:white;">Enter Game</button>
    </div>

    <!-- SCREEN 3: THE GAME -->
    <div id="game-screen" class="screen">
        <h2>Catch the Red Dot!</h2>
        <div id="click-count">Score: <span id="score">0</span></div>
        <div id="game-target" onclick="scorePoint()">TAP ME</div>
        <button onclick="showMenu()" style="background:#555; color:white; font-size:14px;">Quit to Menu</button>
    </div>

<script>
    let score = 0;

    // Register PWA Service Worker for mobile installation
    if ('serviceWorker' in navigator) {
        navigator.serviceWorker.register('service-worker.js');
    }

    // Ad Logic
    function startAd() {
        changeScreen('ad-screen');
        let timeLeft = 5;
        const countdownEl = document.getElementById('countdown');
        const skipBtn = document.getElementById('skip-btn');
        const timerEl = document.getElementById('timer');
        
        skipBtn.style.display = 'none';
        timerEl.style.display = 'block';
        countdownEl.innerText = timeLeft;

        const interval = setInterval(() => {
            timeLeft--;
            countdownEl.innerText = timeLeft;
            if (timeLeft <= 0) {
                clearInterval(interval);
                timerEl.style.display = 'none';
                skipBtn.style.display = 'block'; // Unlocks the game button
            }
        }, 1000);
    }

    // Game Logic
    function startGame() {
        score = 0;
        document.getElementById('score').innerText = score;
        changeScreen('game-screen');
        moveTarget();
    }

    function scorePoint() {
        score++;
        document.getElementById('score').innerText = score;
        moveTarget();
    }

    function moveTarget() {
        const target = document.getElementById('game-target');
        // Randomly bounce the target around the screen space
        const randomColor = '#' + Math.floor(Math.random()*16777215).toString(16);
        target.style.background = randomColor;
        target.style.marginTop = Math.floor(Math.random() * 40 + 10) + 'px';
    }

    function showMenu() {
        changeScreen('menu-screen');
    }

    function changeScreen(screenId) {
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        document.getElementById(screenId).classList.add('active');
    }
</script>
